Algoritmo en Python — los tres métodos 
import math

def doolittle(A):                          # L con diagonal 1
    n = len(A)
    L = [[1.0 if i == j else 0.0 for j in range(n)] for i in range(n)]
    U = [row[:] for row in A]
    for k in range(n):
        for i in range(k + 1, n):
            m = U[i][k] / U[k][k]; L[i][k] = m
            for j in range(k, n): U[i][j] -= m * U[k][j]
    return L, U

def crout(A):                              # U con diagonal 1
    n = len(A)
    L = [[0.0]*n for _ in range(n)]
    U = [[1.0 if i == j else 0.0 for j in range(n)] for i in range(n)]
    for j in range(n):
        for i in range(j, n):
            L[i][j] = A[i][j] - sum(L[i][k]*U[k][j] for k in range(j))
        for i in range(j + 1, n):
            U[j][i] = (A[j][i] - sum(L[j][k]*U[k][i] for k in range(j))) / L[j][j]
    return L, U

def cholesky(A):                           # A simétrica definida positiva
    n = len(A)
    L = [[0.0]*n for _ in range(n)]
    for j in range(n):
        L[j][j] = math.sqrt(A[j][j] - sum(L[j][k]**2 for k in range(j)))
        for i in range(j + 1, n):
            L[i][j] = (A[i][j] - sum(L[i][k]*L[j][k] for k in range(j))) / L[j][j]
    U = [[L[j][i] for j in range(n)] for i in range(n)]   # U = L transpuesta
    return L, U

def resolver(L, U, b):                     # sirve para los tres
    n = len(b); y = [0.0]*n; x = [0.0]*n
    for i in range(n):
        y[i] = (b[i] - sum(L[i][j]*y[j] for j in range(i))) / L[i][i]
    for i in range(n-1, -1, -1):
        x[i] = (y[i] - sum(U[i][j]*x[j] for j in range(i+1, n))) / U[i][i]
    return y, x

A = [[4,2,1],[12,10,5],[-8,8,7]]
L, U = doolittle(A)                        # una sola vez, O(n^3)
for b in ([14,46,26], [20,62,30]):
    print(resolver(L, U, b))               # O(n^2) por vector
